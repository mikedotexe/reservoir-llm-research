import { useEffect, useId, useState } from 'react';
import LocalModelReadiness from './LocalModelReadiness';
import type { SustainedConfig } from './myReservoirTypes';
import { formatSessionDuration, parseSessionDuration, splitSessionDuration, type SessionDurationParts } from './sessionDuration';

function DurationEditor({ value, onChange }: { value: number; onChange: (seconds: number) => void }) {
  const [parts, setParts] = useState(() => splitSessionDuration(value)), id = useId();
  useEffect(() => {
    if (Number.isFinite(value) && value !== parseSessionDuration(parts)) setParts(splitSessionDuration(value));
  }, [value]);
  const invalid = parseSessionDuration(parts) === null;
  const change = (key: keyof SessionDurationParts, text: string) => {
    const next = { ...parts, [key]: text }; setParts(next);
    // Keep an invalid draft invalid for Create; never silently reuse an older duration.
    onChange(parseSessionDuration(next) ?? Number.NaN);
  };
  return <fieldset className="my-reservoir-duration"><legend>Duration limit</legend><div className="my-reservoir-duration-fields">
    {(['hours', 'minutes', 'seconds'] as const).map(key => <label key={key}>{key[0].toUpperCase() + key.slice(1)}<input type="text" inputMode="numeric" pattern="[0-9]{1,2}" required value={parts[key]} aria-invalid={invalid || undefined} aria-describedby={invalid ? id : undefined} onChange={event => change(key, event.target.value)}/></label>)}
  </div>{invalid ? <p id={id} role="alert">Use whole hours, minutes and seconds for a duration from 1 second to 24 hours. Minutes and seconds must be 0–59.</p> : <p className="my-reservoir-small">{formatSessionDuration(value)} · exactly {value} seconds</p>}</fieldset>;
}

    <label className="my-reservoir-source-choice">Reply source<select value={config.language.backend} onChange={event => patch({ language: event.target.value === 'ollama' ? { backend: 'ollama', endpoint: '', model: '', contextTokens: 4096, responseFormat: null } : { backend: 'scripted' } })}><option value="scripted">Deterministic script · synthetic writing</option><option value="ollama">My local Ollama model</option></select></label>
    {config.language.backend === 'scripted' ? <p>Synthetic writing from a deterministic script. ''} placeholder="http://127.0.0.1:11435" spellCheck={false} autoComplete="off" onChange={event => patch({ language: { ...config.language, endpoint: event.target.value } })}/></label><label>Exact model name<input value={config.language.model ?? ''} spellCheck={false} autoComplete="off" onChange={event => patch({ language: { ...config.language, model: event.target.value } })}/></label><label>Context tokens<input type="number" min="512" max="32768" value={config.language.contextTokens ?? 4096} onChange={event => patch({ language: { ...config.language, contextTokens: Number(event.target.value) } })}/></label></div><LocalModelReadiness configuration={config.language} disabled={disabled} generationNote="Create saves local configuration only.       <DurationEditor value={config.maximumDurationSeconds} onChange={maximumDurationSeconds => patch({ maximumDurationSeconds })}/>
      <p>Some storage is reserved for saving, stopping and closing safely. The session can stop before the selected storage limit is full.</p>
      <h3>Numerical controls and observation policy</h3><div className="my-reservoir-config-grid"><label>Leak · new-state mix<input type="number" min="0" max="1" step=".01" value={config.leak} onChange={event => patch({ leak: Number(event.target.value) })}/></label><label>Seeded noise amplitude<input type="number" min="0" max=".2" step=".005" value={config.noiseAmplitude} onChange={event => patch({ noiseAmplitude: Number(event.target.value) })}/></label><label>Initial sensory retention<input type="number" min=".82" max=".995" step=".005" value={config.initialRetention} onChange={event => patch({ initialRetention: Number(event.target.value) })}/></label><label>Sensory retention regulation<select value={String(config.regulationEnabled)} onChange={event => patch({ regulationEnabled: event.target.value === 'true' })}><option value="false">Fixed retention</option><option value="true">Regulate separate sensory fill</option></select></label></div><p>Each writing opportunity includes all 32 reservoir coordinates, their RMS, L2 and observed change, plus separate sensory measurements.     </details>
    <p className="my-reservoir-small">This app runs while open or minimized, with no background service. Create prepares local state; Start is a separate action.</p>
